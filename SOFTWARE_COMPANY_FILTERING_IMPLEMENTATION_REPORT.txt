### 2.2 Software Company Filtering Logic (NEW):
```typescript
const softwareCompanyFilteredPosters = useMemo(() => {
  // Only apply filtering if Software Company buttons are active and a category is selected
  if (!shouldShowSoftwareButtons || !selectedSoftwareCategory) {
    return allTemplates;
  }

  console.log('?? [SOFTWARE COMPANY FILTERING] Starting filtering process');
  console.log(`?? [SOFTWARE COMPANY FILTERING] Total posters before filtering: ${allTemplates.length}`);
  console.log(`?? [SOFTWARE COMPANY FILTERING] Selected category: ${selectedSoftwareCategory}`);
  
  let basePosters = allTemplates;

  // Apply category filtering if a specific category is selected
  if (selectedSoftwareCategory !== 'all') {
    const selectedCategoryButton = softwareCategoryButtons.find(btn => btn.id === selectedSoftwareCategory);
    if (selectedCategoryButton && selectedCategoryButton.tags.length > 0) {
      console.log(`?? [SOFTWARE COMPANY FILTERING] Filtering by category: ${selectedCategoryButton.name}`);
      console.log(`?? [SOFTWARE COMPANY FILTERING] Category tags: [${selectedCategoryButton.tags.join(', ')}]`);
      
      const postersBeforeCategoryFilter = basePosters.length;
      basePosters = basePosters.filter(poster => {
        const posterTags = poster.tags || [];
        const posterName = poster.name.toLowerCase();
        
        // Check if poster matches any of the selected category's tags
        const matchesTag = selectedCategoryButton.tags.some(tag => {
          const tagLower = tag.toLowerCase();
          return posterTags.some(posterTag => 
            posterTag.toLowerCase() === tagLower ||
            posterTag.toLowerCase().includes(tagLower) ||
            tagLower.includes(posterTag.toLowerCase())
          ) || posterName.includes(tagLower);
        });
        
        if (matchesTag) {
          console.log(`?? [SOFTWARE COMPANY FILTERING] Poster "${poster.name}" matched - Tags: [${posterTags.join(', ')}]`);
        } else {
          console.log(`?? [SOFTWARE COMPANY FILTERING] Poster "${poster.name}" did NOT match - Tags: [${posterTags.join(', ')}]`);
        }
        
        return matchesTag;
      });
      
      console.log(`?? [SOFTWARE COMPANY FILTERING] Posters after category filtering: ${basePosters.length} (removed ${postersBeforeCategoryFilter - basePosters.length})`);
    }
  } else {
    console.log(`?? [SOFTWARE COMPANY FILTERING] No category filter applied (selected: 'all')`);
  }

  console.log(`?? [SOFTWARE COMPANY FILTERING COMPLETE] Final filtered posters count: ${basePosters.length}`);
  return basePosters;
}, [allTemplates, shouldShowSoftwareButtons, selectedSoftwareCategory, softwareCategoryButtons]);
```

    return matches;
  });

  if (languageFilteredForSoftware.length === 0) {
    console.log(`[SOFTWARE COMPANY] No templates found for language: ${selectedLanguage}`);
    
    // SAFE FALLBACK LOGIC
    // If we have a current poster, return it as only result
    if (currentPoster && currentPoster.id !== 'loading' && !currentPoster.id.startsWith('category_')) {
      console.log(`[FALLBACK] Returning current poster as only result: ${currentPoster.name}`);
      return [currentPoster];
    }
    
    console.log(`\ud83c\udf89 [FILTERING COMPLETE] Final filtered templates count: 0`);
    return [];
  }

  const limitedTemplates = languageFilteredForSoftware.slice(0, 6);
  console.log(`[SOFTWARE COMPANY] After language filter: ${languageFilteredForSoftware.length}, Showing: ${Math.min(languageFilteredForSoftware.length, 6)} templates (language: ${selectedLanguage})`);
  return limitedTemplates;
