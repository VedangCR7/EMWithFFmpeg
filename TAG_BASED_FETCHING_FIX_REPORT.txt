## 2. Technical Implementation:

### 2.1 API Call Pattern:
```typescript
const fetchTemplatesByTags = useCallback(async (tags: string[]) => {
  if (!tags.length) return;
  
  setIsLoadingServiceFilter(prev => ({ ...prev, tagfetch: true }));

  try {
    // Use businessCategoryPostersApi like Event Planner does
    const response = await businessCategoryPostersApi.getPostersByCategory('software company', 500);

    if (response?.success && Array.isArray(response.data?.posters)) {
      // Convert to Template format
      const allTemplates: Template[] = response.data.posters.map((poster: any) => ({
        id: poster.id,
        name: poster.name || poster.title || 'Untitled',
        category: poster.category || 'Software Company',
        imageUrl: poster.imageUrl || poster.image || poster.thumbnail,
        thumbnailUrl: poster.thumbnailUrl || poster.thumbnail || poster.imageUrl || poster.image,
        tags: Array.isArray(poster.tags) ? poster.tags : [],
        languages: Array.isArray(poster.languages) ? poster.languages : ['English'],
        isPremium: poster.isPremium || false,
        aspectRatio: poster.aspectRatio || '1:1',
        createdAt: poster.createdAt || new Date().toISOString(),
        updatedAt: poster.updatedAt || new Date().toISOString(),
      }));

      // Filter templates by requested tags (limit to 6)
      const filteredTemplates = allTemplates
        .filter(template => {
          const templateTags = template.tags || [];
          const templateName = template.name.toLowerCase();
          
          return tags.some(tag => {
            const tagLower = tag.toLowerCase();
            return templateTags.some(posterTag => 
              posterTag.toLowerCase() === tagLower ||
              posterTag.toLowerCase().includes(tagLower) ||
              tagLower.includes(posterTag.toLowerCase())
            ) || templateName.includes(tagLower);
          });
        })
        .slice(0, 6); // Limit to 6 templates

      setTagBasedTemplates(filteredTemplates);
      return filteredTemplates;
    }
  } catch (error) {
    console.error(`?? [TAG FETCH] Error fetching templates by tags:`, error);
  } finally {
    setIsLoadingServiceFilter(prev => ({ ...prev, tagfetch: false }));
  }

  // Fallback: return empty array if fetch fails
  setTagBasedTemplates([]);
  return [];
}, []);
```

### 2.2 Tag Filtering Logic:
