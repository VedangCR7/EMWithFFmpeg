```typescript
const shouldShowSoftwareButtons = useMemo(() => {
  // Only validate if we have Software Company category detection
  if (!isSoftwareCompanyCategory) return false;
  
  // My Business Flow: Allow buttons (IndustryCategoryScreen handles this)
  if (flowSource === "MyBusiness") return true;
  
  // HomeScreen Flow: Allow buttons only if category name is Software Company
  if (flowSource === "HomeScreen") {
    const homeScreenCategory = (categoryName || '').trim().toLowerCase();
    const softwareCompanyVariations = [
      'software companies',
      'software company',
      'software-companies',
      'software-company',
      'softwarecompanies',
      'softwarecompany'
    ];
    return softwareCompanyVariations.some(variation => homeScreenCategory.includes(variation));
  }
  
  // Other flows: Do not allow buttons
  return false;
}, [flowSource, isSoftwareCompanyCategory, categoryName]);
```
