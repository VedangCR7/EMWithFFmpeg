router.post('/register', async (req: Request, res: Response) => {
  // 1. Input Extraction
  const { deviceId, name, email, password, phone, ... } = req.body;
  
  // 2. Device ID Generation (if missing)
  let finalDeviceId = deviceId || generateDeviceId();
  
  // 3. Email Uniqueness Validation
  if (email && password) {
    const existingEmailUser = await prisma.mobileUser.findUnique({
      where: { email }
    });
    if (existingEmailUser) {
      return res.status(409).json({
        success: false,
        error: 'User with this email already exists'
      });
    }
  }
  
  // 4. Password Hashing
  const hashedPassword = await bcrypt.hash(password, 12);
  
  // 5. Multi-Record Creation
  const mobileUser = await prisma.mobileUser.create({...});
  const customer = await prisma.customer.create({...});
  const businessProfile = await prisma.businessProfile.create({...});
  
  // 6. JWT Token Generation
  const token = jwt.sign({
    id: mobileUser.id,
    email: mobileUser.email,
    userType: 'MOBILE_USER'
  }, process.env.JWT_SECRET, { expiresIn: '365d' });
  
  // 7. Response
  res.status(201).json({
    success: true,
    data: { user: mobileUser, token, businessProfile }
  });
});
